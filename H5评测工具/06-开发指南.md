## 1. 开发环境搭建

### 1.1 前端环境

#### **系统要求**
- 操作系统：Windows 10/11, macOS 10.15+, Ubuntu 18.04+
- 内存：≥8GB
- 硬盘：≥50GB可用空间

#### **软件安装**

**1. 安装Node.js（前端构建工具）**
```bash

https://nodejs.org/dist/v18.17.0/node-v18.17.0-x64.msi

# macOS（使用Homebrew）
brew install node@18

# Ubuntu/Debian
curl -fsSL https://deb.nodesource.com/setup_18.x | sudo -E bash -
sudo apt-get install -y nodejs

# 验证安装
node -v  # 应显示 v18.x.x
npm -v   # 应显示 9.x.x
```

**2. 安装Vue CLI（可选，用于本地开发）**
```bash
npm install -g @vue/cli
vue --version  # 应显示 5.x.x
```

**3. 安装微信开发者工具**
- 下载地址：https://developers.weixin.qq.com/miniprogram/dev/devtools/download.html

## 3. 核心代码实现

### 3.1 前端核心代码

#### **API接口封装（src/api/index.js）**
```javascript
import axios from 'axios';

const api = axios.create({
  baseURL: process.env.VUE_APP_API_BASE || 'http://localhost:3000/api/v1',
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
});

// 请求拦截器
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
}, error => {
  return Promise.reject(error);
});

// 响应拦截器
api.interceptors.response.use(response => {
  return response.data;
}, error => {
  if (error.response.status === 401) {
    // Token过期，重新授权
    localStorage.removeItem('token');
    window.location.href = '/';
  }
  return Promise.reject(error);
});

export default api;

// 题目相关接口
export const questionApi = {
  getQuestions(grade) {
    return api.get('/questions', { params: { grade } });
  }
};

// 提交相关接口
export const submitApi = {
  submitAnswers(data) {
    return api.post('/submit', data);
  }
};

// 结果相关接口
export const resultApi = {
  getResult(resultId) {
    return api.get(`/result/${resultId}`);
  },
  generatePoster(resultId) {
    return api.post('/poster', { resultId });
  }
};
```

#### **微信授权逻辑（src/utils/auth.js）**
```javascript
import wx from 'weixin-js-sdk';

export const wechatAuth = {
  /**
   * 初始化微信配置
   */
  initWechatConfig() {
    return new Promise((resolve, reject) => {
      // 从后端获取微信配置
      fetch('/api/wechat/config')
        .then(res => res.json())
        .then(config => {
          wx.config({
            appId: config.appId,
            timestamp: config.timestamp,
            nonceStr: config.nonceStr,
            signature: config.signature,
            jsApiList: [
              'updateAppMessageShareData',
              'updateTimelineShareData',
              'onMenuShareTimeline',
              'onMenuShareAppMessage'
            ]
          });

          wx.ready(() => {
            resolve();
          });

          wx.error((err) => {
            reject(err);
          });
        });
    });
  },

  /**
   * 获取用户openid（通过后端接口）
   */
  getOpenId(code) {
    return fetch(`/api/wechat/openid?code=${code}`)
      .then(res => res.json());
  }
};
```

```javascript
const express = require('express');
const router = express.Router();
const questionCtrl = require('../controllers/questionCtrl');

/**
 * 获取题目列表
 * GET /api/v1/questions?grade=2
 */
router.get('/', async (req, res) => {
  try {
    const { grade } = req.query;
    
    if (!grade) {
      return res.json({ code: 1001, message: '缺少grade参数' });
    }

    const questions = await questionCtrl.getQuestions(grade);
    
    res.json({
      code: 0,
      message: 'success',

  } catch (error) {
    console.error('获取题目失败:', error);
    res.json({ code: 5000, message: '服务器内部错误' });
  }
});

module.exports = router;
```

#### **提交控制器（controllers/submitCtrl.js）**
```javascript
const scoreService = require('../services/scoreService');
const posterService = require('../services/posterService');
const pool = require('../config/database');

class SubmitController {
  /**
   * 提交测评答案
   */
  async submit(req, res) {
    try {
      const { openid, answers } = req.body;

      // 参数校验
      if (!openid || !answers || answers.length !== 25) {
        return res.json({ code: 1001, message: '参数错误' });
      }

      // 检查测评次数（7天内只能测一次）
      const [lastTest] = await pool.query(
        'SELECT create_time FROM results WHERE user_id = ? ORDER BY create_time DESC LIMIT 1',
        [openid]
      );

      if (lastTest.length > 0) {
        const daysSinceLastTest = (Date.now() - lastTest[0].create_time) / (1000 * 60 * 60 * 24);
        if (daysSinceLastTest < 7) {
          return res.json({ code: 1004, message: '7天内只能测评一次' });
        }
      }

      // 获取题目列表
      const [questions] = await pool.query(
        'SELECT * FROM questions WHERE status = 1 ORDER BY RAND() LIMIT 25'
      );

      // 计算得分
      const scores = scoreService.calculateScores(answers, questions);

      // 匹配课程
      const matchCourse = scoreService.matchCourse(scores);

      // 生成雷达图
      const radarImage = await posterService.generateRadarChart(scores);

      // 保存结果
      const [result] = await pool.query(
        `INSERT INTO results (user_id, answers, scores, radar_image, match_course, create_time)
         VALUES (?, ?, ?, ?, ?, ?)`,
        [
          openid,
          JSON.stringify(answers),
          JSON.stringify(scores),
          radarImage,
          matchCourse,
          Date.now()
        ]
      );

      res.json({
        code: 0,
        message: '测评完成',
        data: {
          resultId: result.insertId,
          scores,
          radarImage,
          matchCourse
        }
      });
    } catch (error) {
      console.error('提交测评失败:', error);
      res.json({ code: 5000, message: '服务器内部错误' });
    }
  }
}

module.exports = new SubmitController();
```

## 4. 数据库初始化

### 4.1 初始化脚本（scripts/initdb.js）

```javascript
const pool = require('../config/database');
const fs = require('fs');
const path = require('path');

async function initDB() {
  try {
    console.log('开始初始化数据库...');

    // 读取SQL文件
    const sqlPath = path.join(__dirname, '../docs/数据库设计.sql');
    const sql = fs.readFileSync(sqlPath, 'utf8');

    // 执行SQL
    const statements = sql.split(';').filter(s => s.trim());
    for (const statement of statements) {
      await pool.query(statement);
    }

    console.log('数据库初始化成功！');
    process.exit(0);
  } catch (error) {
    console.error('数据库初始化失败:', error);
    process.exit(1);
  }
}

initDB();
```

## 5. 开发工作流

### 5.1 Git分支管理

```
main (生产分支)
  ↓
develop (开发分支)
  ↓
feature/xxx (功能分支)
  ↓
hotfix/xxx (紧急修复分支)
```

**分支命名规范**：
- 功能分支：`feature/模块名-功能名`（如`feature/quiz-answer`）
- 修复分支：`fix/问题描述`（如`fix/score-calc-error`）
- 热修复：`hotfix/问题描述`（如`hotfix/crash-on-ios`）

### 5.2 提交规范

使用**Conventional Commits**规范：

```
<type>(<scope>): <subject>

<body>

<footer>
```

**type类型**：
- `feat`: 新功能
- `fix`: 修复bug
- `docs`: 文档更新
- `style`: 代码格式调整（不影响功能）
- `refactor`: 重构（既不是新功能也不是修复）
- `perf`: 性能优化
- `test`: 测试相关
- `chore`: 构建过程或辅助工具的变动

**示例**：
```bash
feat(quiz): 添加答题进度显示功能

- 在答题页面顶部添加进度条
- 显示当前题号和总题数
- 进度条动画效果

Closes #123
```

### 5.3 代码审查流程

1. 开发者完成功能开发
2. 本地测试通过
3. 提交Pull Request到`develop`分支
4. 指定审查人（至少1人）
5. 代码审查（检查代码质量、规范、安全性）
6. 审查通过后合并
7. 删除功能分支

## 6. 常见问题与解决方案

### 6.1 前端常见问题

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 微信JS-SDK配置失败 | 签名错误/URL不匹配 | 检查后端签名算法，确保URL是动态获取的 |
| 跨域问题 | CORS策略限制 | 后端添加CORS中间件，设置`Access-Control-Allow-Origin` |
| 图片加载慢 | 图片体积大/未压缩 | 使用TinyPNG压缩，上传到CDN |
| 雷达图不显示 | ECharts未正确初始化 | 确保在`mounted`钩子中初始化，检查容器尺寸 |

### 6.2 后端常见问题

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 数据库连接池耗尽 | 连接未释放 | 使用`try-catch-finally`确保连接释放 |
| SQL注入风险 | 未使用参数化查询 | 使用`?`占位符，避免字符串拼接 |
| Redis连接失败 | 配置错误/服务未启动 | 检查Redis配置，确保服务正常运行 |
| JWT验证失败 | Token过期/签名错误 | 检查Token有效期，确保密钥一致 |

## 7. 性能优化建议

### 7.1 前端优化

1. **代码分割**：使用动态`import()`懒加载路由组件
2. **图片优化**：使用WebP格式，添加`loading="lazy"`属性
3. **缓存策略**：使用`localStorage`缓存题目数据（有效期1小时）
4. **CDN加速**：静态资源（JS/CSS/图片）上传到CDN

### 7.2 后端优化

1. **数据库索引**：为常用查询字段添加索引（`openid`, `grade`, `create_time`）
2. **Redis缓存**：缓存热门题目，减少数据库查询
3. **Gzip压缩**：启用Gzip压缩响应数据
4. **连接池**：使用数据库连接池，避免频繁创建连接
