```javascript
        state.answers.push(answer);
      }

      state.result = result;
    }
  },

        body: JSON.stringify({
          openid: state.user.openid,
          answers: state.answers
        })
      });
      return res.json();
    }
```

### 5.2 核心代码实现

#### **评分算法实现（scoreService.js）**
```javascript
// services/scoreService.js
class ScoreService {
  /**
   * 计算五维得分
   * @param {Array} answers - 答题记录 [{qid, option}]
   * @param {Array} questions - 题目列表
   * @returns {Object} 五维得分 {logic, creativity, concept, focus, collaboration}
   */
  calculateScores(answers, questions) {
    const dimensions = {
      logic: [],
      creativity: [],
      concept: [],
      focus: [],
      collaboration: []
    };

    // 遍历答案，累加各维度得分
    answers.forEach(answer => {
      const question = questions.find(q => q.id === answer.qid);
      if (!question) return;

      const scores = question[`scores_${answer.option.toLowerCase()}`];
      Object.keys(scores).forEach(dim => {
        if (dimensions[dim]) {
          dimensions[dim].push(scores[dim]);
        }
      });
    });

    // 计算平均分并归一化（0-4分 → 0-100分）
    const result = {};
    Object.keys(dimensions).forEach(dim => {
      const scores = dimensions[dim];
      if (scores.length === 0) {
        result[dim] = 0;
      } else {
        const avg = scores.reduce((a, b) => a + b, 0) / scores.length;
        result[dim] = Math.round(avg * 25); // 0-4 → 0-100
      }
    });

    return result;
  }

  /**
   * 匹配推荐课程
   * @param {Object} scores - 五维得分
   * @returns {String} 推荐课程名称
   */
  matchCourse(scores) {
    const { logic, creativity, concept, focus, collaboration } = scores;

    // 规则1：逻辑思维高 → Python编程
    if (logic >= 80) {
      return 'Python编程基础';
    }

    // 规则2：创意实现高 → Scratch游戏设计
    if (creativity >= 80) {
      return 'Scratch游戏设计';
    }

    // 规则3：概念理解高 → C++基础
    if (concept >= 80) {
      return 'C++编程基础';
    }

    // 规则4：综合均衡 → 可视化编程
    const avg = (logic + creativity + concept + focus + collaboration) / 5;
    if (avg >= 60) {
      return '可视化编程入门';
    }

    // 默认推荐
    return '编程思维启蒙';
  }
}

module.exports = new ScoreService();
```

#### **提交控制器实现（submitCtrl.js）**
```javascript
// controllers/submitCtrl.js
const ScoreService = require('../services/scoreService');
const PosterService = require('../services/posterService');

class SubmitController {
  /**
   * 提交测评答案
   */
  async submit(req, res) {
    try {
      const { openid, answers } = req.body;

      // 参数校验
      if (!openid || !answers || answers.length !== 25) {
        return res.json({ code: 1001, message: '参数错误' });
      }

      // 检查测评次数（7天内只能测一次）
      const lastTest = await db.query(
        'SELECT create_time FROM results WHERE user_id = ? ORDER BY create_time DESC LIMIT 1',
        [openid]
      );
      if (lastTest.length > 0) {
        const daysSinceLastTest = (Date.now() - lastTest[0].create_time) / (1000 * 60 * 60 * 24);
        if (daysSinceLastTest < 7) {
          return res.json({ code: 1004, message: '7天内只能测评一次' });
        }
      }

      // 获取题目列表
      const questions = await db.query(
        'SELECT * FROM questions WHERE status = 1 ORDER BY RAND() LIMIT 25'
      );

      // 计算得分
      const scores = ScoreService.calculateScores(answers, questions);

      // 匹配课程
      const matchCourse = ScoreService.matchCourse(scores);

      // 生成雷达图
      const radarImage = await PosterService.generateRadarChart(scores);

      // 生成报告
      const reportUrl = await PosterService.generateReport(scores, matchCourse);

      // 保存结果
      const resultId = await db.insert('results', {
        user_id: openid,
        answers: JSON.stringify(answers),
        scores: JSON.stringify(scores),
        radar_image: radarImage,
        report_url: reportUrl,
        match_course: matchCourse,
        create_time: Date.now()
      });

      // 返回结果
      res.json({
        code: 0,
        message: '测评完成',
        data: {
          resultId,
          scores,
          reportUrl,
          matchCourse
        }
      });
    } catch (error) {
      console.error('提交测评失败:', error);
      res.json({ code: 5000, message: '服务器内部错误' });
    }
  }
}

module.exports = new SubmitController();
```

## 6. 算法详细设计

### 6.1 评分算法流程图

```
开始
  ↓
初始化五维数组（logic, creativity, concept, focus, collaboration）
  ↓
遍历用户答案（25条）
  ↓
根据qid查找题目
  ↓
根据选项（A/B/C/D）获取对应得分
  ↓
将得分追加到对应维度数组
  ↓
计算各维度平均分（0-4分）
  ↓
归一化为百分制（0-100分）
  ↓
返回五维得分JSON
```

```javascript
// services/posterService.js
class PosterService {
  /**
   * 生成分享海报
   * @param {Object} scores - 五维得分
   * @param {String} qrcodeUrl - 二维码URL
   * @returns {String} 海报URL
   */
  async generatePoster(scores, qrcodeUrl) {
    // 创建Canvas
    const canvas = createCanvas(750, 1334);
    const ctx = canvas.getContext('2d');

    // 1. 绘制背景
    const bg = await loadImage('assets/poster_bg.png');

    ctx.font = 'bold 48px sans-serif';
    ctx.fillStyle = '#333';
    ctx.fillText('我的编程天赋测评结果', 100, 100);

    // 3. 绘制雷达图
    const radarImage = await this.generateRadarChart(scores);
    ctx.drawImage(radarImage, 150, 200, 450, 450);

    // 4. 绘制二维码
    const qrcode = await loadImage(qrcodeUrl);
    ctx.drawImage(qrcode, 275, 750, 200, 200);

    // 5. 绘制文案
    ctx.font = '28px sans-serif';
    ctx.fillStyle = '#666';
    ctx.fillText('扫码查看你的编程天赋', 200, 1000);

    // 6. 导出为图片
    const posterBuffer = canvas.toBuffer('image/png');
    const posterUrl = await uploadToCDN(posterBuffer);

    return posterUrl;
  }
}
```

### 7.1 数据加密

**用户openid加密存储**：
```javascript
const crypto = require('crypto');
const algorithm = 'aes-256-gcm';
const secretKey = process.env.ENCRYPT_KEY;

function encrypt(text) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv(algorithm, secretKey, iv);
  cipher.update(text, 'utf8', 'hex');
  const encrypted = cipher.final('hex');
  const authTag = cipher.getAuthTag();
  return { iv: iv.toString('hex'), encrypted, authTag: authTag.toString('hex') };
}
```

### 7.2 防SQL注入

**使用参数化查询**：
```javascript
// ❌ 错误示例（SQL注入风险）
const sql = `SELECT * FROM users WHERE openid = '${openid}'`;

// ✅ 正确示例（参数化查询）
const sql = 'SELECT * FROM users WHERE openid = ?';
const result = await db.query(sql, [openid]);
```

### 8.2 Redis缓存策略

```javascript
// 缓存题目列表（按年级缓存，过期时间1小时）
async function getQuestions(grade) {
  const cacheKey = `questions:grade:${grade}`;
  let questions = await redis.get(cacheKey);
  
  if (!questions) {
    questions = await db.query('SELECT * FROM questions WHERE grade = ? AND status = 1', [grade]);
    await redis.setex(cacheKey, 3600, JSON.stringify(questions));
  } else {
    questions = JSON.parse(questions);
  }
  
  return questions;
}
```
